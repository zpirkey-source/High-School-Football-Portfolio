MODULE
GameClock · excerpt
function GameClock:GetDisplay(): string
    local minutes = math.floor(self.Remaining / 60)
    local seconds = self.Remaining % 60
    return string.format("%d:%02d", minutes, seconds)
end

function GameClock:NextQuarter()
    self:Pause()
    if self.Quarter >= self.QuarterCount then
        return false
    end
    self.Quarter += 1
    self.Remaining = self.SecondsPerQuarter
    self.Changed:Fire(self:GetDisplay(), self.Quarter, false)
    return true
end

ScoreService · excerpt
local validAwards = {
    Touchdown = 6,
    FieldGoal = 3,
    Safety = 2,
    ExtraPoint = 1,
    TwoPoint = 2,
}

local function award(teamName, playType)
    local points = validAwards[playType]
    if not points or scores[teamName] == nil then
        return false, "Invalid scoring event"
    end

    scores[teamName] += points
    ScoreChanged:FireAllClients(scores.Home, scores.Away)
    return true
end
